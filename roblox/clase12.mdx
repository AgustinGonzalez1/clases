## Paso 3: El script

Va en un **LocalScript** dentro del Tool.

```lua
local tool = script.Parent                            -- el Tool (la pistola)
local jugador = game.Players.LocalPlayer              -- el jugador que ejecuta este script
local personaje = jugador.Character                   -- personaje fisico del jugador
local canon = tool:WaitForChild("Canon")              -- espera a que el Canon este listo
local TweenService = game:GetService("TweenService")  -- para animar la bala

local gui = jugador.PlayerGui:WaitForChild("ScreenGui")
local texto = gui:WaitForChild("TextLabel")

local disparos = 0

tool.Activated:Connect(function()                     -- se dispara al hacer click con el Tool
    disparos = disparos + 1
    texto.Text = "Disparos: " .. disparos

    local camara = workspace.CurrentCamera
    local direccion = camara.CFrame.LookVector         -- hacia donde mira la camara

    local bala = Instance.new("Part")                  -- crea el proyectil
    bala.Size = Vector3.new(0.3, 0.3, 0.3)
    bala.Color = Color3.fromRGB(255, 255, 0)
    bala.Anchored = true
    bala.CanCollide = false
    bala.CFrame = CFrame.new(canon.Position)           -- sale desde el canon
    bala.Parent = workspace

    local destino = canon.Position + direccion * 200   -- 200 studs hacia adelante
    local tween = TweenService:Create(bala, TweenInfo.new(1), {Position = destino})
    tween:Play()

    game:GetService("Debris"):AddItem(bala, 3)         -- destruye la bala en 3 segundos
end)
```
